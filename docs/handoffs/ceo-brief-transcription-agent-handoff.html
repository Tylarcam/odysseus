# Waterfall entry — never raises; returns normalized dict
async def transcribe_video_url(url: str) -> Dict[str, Any]:
    url = (url or "").strip()
    if not url.startswith(("http://", "https://")):
        return _normalize(success=False, source_url=url,
                          error="Provide a full http(s) video URL")
    result = await _try_captions(url)          # youtube_transcript_api FULL text
    if result:
        return result
    result = await _try_aether_server(url)     # POST {AETHER}/api/transcribe-url
    if result:
        return result
    return _normalize(success=False, source_url=url, error="All transcript paths failed…")
